cmake_minimum_required(VERSION 3.22.1)
project(yolo_jni CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_SHARED_LINKER_FLAGS "${CMAKE_SHARED_LINKER_FLAGS} -Wl,-z,max-page-size=16384")

# CI supplies a hash-verified llama.cpp d6cf0b288 archive and Vulkan headers.
set(LLAMA_SRC ${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_STANDALONE OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_UI OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN ON CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
set(Vulkan_INCLUDE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/vulkan-headers/include CACHE PATH "" FORCE)
set(Vulkan_GLSLC_EXECUTABLE /usr/bin/glslc CACHE FILEPATH "" FORCE)
if (NOT DEFINED ANDROID_HOST_TAG)
    set(ANDROID_HOST_TAG linux-x86_64)
endif()
set(Vulkan_LIBRARY
    ${ANDROID_NDK}/toolchains/llvm/prebuilt/${ANDROID_HOST_TAG}/sysroot/usr/lib/aarch64-linux-android/30/libvulkan.so
    CACHE FILEPATH "" FORCE)
set(SPIRV-Headers_DIR /usr/share/cmake/SPIRV-Headers CACHE PATH "" FORCE)
file(WRITE ${CMAKE_BINARY_DIR}/vk-host-toolchain.cmake "
set(CMAKE_BUILD_TYPE Release)
set(CMAKE_C_COMPILER /usr/bin/cc)
set(CMAKE_CXX_COMPILER /usr/bin/c++)
set(CMAKE_MAKE_PROGRAM \"${CMAKE_MAKE_PROGRAM}\" CACHE FILEPATH \"\" FORCE)
set(CMAKE_FIND_ROOT_PATH_MODE_PROGRAM NEVER)
set(CMAKE_FIND_ROOT_PATH_MODE_LIBRARY NEVER)
set(CMAKE_FIND_ROOT_PATH_MODE_INCLUDE NEVER)
")
set(GGML_VULKAN_SHADERS_GEN_TOOLCHAIN ${CMAKE_BINARY_DIR}/vk-host-toolchain.cmake CACHE FILEPATH "" FORCE)
add_subdirectory(${LLAMA_SRC} ${CMAKE_CURRENT_BINARY_DIR}/llama.cpp)
# This pinned version predates LLAMA_BUILD_MTMD; request its library explicitly.
add_subdirectory(${LLAMA_SRC}/tools/mtmd ${CMAKE_CURRENT_BINARY_DIR}/mtmd EXCLUDE_FROM_ALL)
add_library(vlm_jni SHARED vlm_jni.cpp)
target_include_directories(vlm_jni PRIVATE
    ${LLAMA_SRC}/include ${LLAMA_SRC}/ggml/include ${LLAMA_SRC}/tools/mtmd)
find_library(android_log log)
target_link_libraries(vlm_jni PRIVATE mtmd llama ${android_log} jnigraphics android)

set(NCNN_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/ncnn)
set(ncnn_DIR ${NCNN_ROOT}/${ANDROID_ABI}/lib/cmake/ncnn)
find_package(ncnn REQUIRED)
find_library(log_lib log)
add_library(yolo_jni SHARED yolo_jni.cpp)
target_compile_features(yolo_jni PRIVATE cxx_std_17)
target_link_options(yolo_jni PRIVATE "-Wl,-z,max-page-size=16384" -fopenmp -static-openmp)
target_link_libraries(yolo_jni PRIVATE ncnn ${log_lib} jnigraphics android)
